import os, sys

ROOTS = [r'D:\workbuddyspace', r'D:\reasonix']
EXCLUDE_DIRS = {'node_modules', '.venv', 'venv', 'site-packages', '__pycache__',
                '.git', 'qlib_data', 'databases', 'data', 'redis_data', 'tdx_csv', 'cache'}
TEXT_EXT = {'.py', '.bat', '.ps1', '.json', '.toml', '.sh', '.txt', '.cfg',
            '.ini', '.yaml', '.yml', '.md', '.url'}

# 顺序无关:三种写法的字节序列互不包含
PAIRS = [
    ('D:\\\\workbuddyspace', 'D:\\\\zcodesj\\\\workbuddyspace'),
    ('d:\\\\workbuddyspace', 'd:\\\\zcodesj\\\\workbuddyspace'),
    ('D:\\workbuddyspace', 'D:\\zcodesj\\workbuddyspace'),
    ('d:\\workbuddyspace', 'd:\\zcodesj\\workbuddyspace'),
    ('D:/workbuddyspace', 'D:/zcodesj/workbuddyspace'),
    ('d:/workbuddyspace', 'd:/zcodesj/workbuddyspace'),
    ('D:\\\\reasonix', 'D:\\\\zcodesj\\\\reasonix'),
    ('d:\\\\reasonix', 'd:\\\\zcodesj\\\\reasonix'),
    ('D:\\reasonix', 'D:\\zcodesj\\reasonix'),
    ('d:\\reasonix', 'd:\\zcodesj\\reasonix'),
    ('D:/reasonix', 'D:/zcodesj/reasonix'),
    ('d:/reasonix', 'd:/zcodesj/reasonix'),
]

changed, skipped_bin = [], 0
for root in ROOTS:
    for dirpath, dirnames, filenames in os.walk(root):
        rel = os.path.relpath(dirpath, root)
        top = rel.split(os.sep)[0]
        if top.startswith('2026-'):      # 近期会话快照,不动
            dirnames[:] = []
            continue
        dirnames[:] = [d for d in dirnames if d not in EXCLUDE_DIRS]
        for fn in filenames:
            if os.path.splitext(fn)[1].lower() not in TEXT_EXT:
                continue
            p = os.path.join(dirpath, fn)
            try:
                raw = open(p, 'rb').read()
            except OSError:
                continue
            if b'\x00' in raw[:8192]:     # 二进制跳过
                skipped_bin += 1
                continue
            new = raw
            for old, rep in PAIRS:
                new = new.replace(old.encode(), rep.encode())
            if new != raw:
                open(p, 'wb').write(new)
                changed.append(p)

with open(r'D:\zcodesj\migration_manifest_20260809.txt', 'w', encoding='utf-8') as f:
    f.write('\n'.join(changed))
print(f'changed={len(changed)} skipped_bin={skipped_bin}')

"""Batch-update northbound_holding with latest quarterly data from westock CLI.

Source : npx -y westock-data-skillhub@1.0.5 fund north-holding <codes> --raw
Target : D:\\workbuddyspace\\databases\\fundamental_data.db  (northbound_holding)
"""
import json
import sqlite3
import subprocess
import sys
import time

DB = r"D:\workbuddyspace\databases\fundamental_data.db"
BATCH = 200
SLEEP = 0.2
NPX = "npx.cmd"
PKG = "westock-data-skillhub@1.0.5"


def to_westock(ts_code: str) -> str:
    num, sfx = ts_code.split(".")
    return {"SH": "sh", "SZ": "sz", "BJ": "bj"}[sfx] + num


def to_tscode(wcode: str) -> str:
    pfx, num = wcode[:2], wcode[2:]
    return num + {"sh": ".SH", "sz": ".SZ", "bj": ".BJ"}[pfx]


def fetch_batch(codes):
    """Return (items, error_str). items may be None on hard failure."""
    cmd = [NPX, "-y", PKG, "fund", "north-holding", ",".join(codes), "--raw"]
    try:
        r = subprocess.run(cmd, capture_output=True, text=True,
                           encoding="utf-8", timeout=300)
    except Exception as e:  # noqa: BLE001
        return None, f"subprocess error: {e}"
    if r.returncode != 0:
        return None, f"rc={r.returncode} stderr={r.stderr[:300]}"
    try:
        return json.loads(r.stdout), None
    except Exception as e:  # noqa: BLE001
        return None, f"json parse error: {e}; head={r.stdout[:200]}"


def main():
    conn = sqlite3.connect(DB)
    cur = conn.cursor()
    cur.execute("SELECT DISTINCT ts_code FROM northbound_holding ORDER BY ts_code")
    ts_codes = [r[0] for r in cur.fetchall()]
    wcodes = [to_westock(c) for c in ts_codes]
    w2ts = dict(zip(wcodes, ts_codes))
    print(f"total codes: {len(wcodes)}", flush=True)

    # existing (ts_code, trade_date) keys for upsert decision
    cur.execute("SELECT ts_code, trade_date FROM northbound_holding")
    existing = set(cur.fetchall())

    n_insert = n_update = 0
    ok_codes = set()
    fail_codes = []      # (code, reason)
    empty_codes = []     # codes absent from response

    batches = [wcodes[i:i + BATCH] for i in range(0, len(wcodes), BATCH)]
    for bi, batch in enumerate(batches, 1):
        items, err = fetch_batch(batch)
        if items is None:
            print(f"[batch {bi}/{len(batches)}] FAILED: {err}", flush=True)
            fail_codes.extend((c, err) for c in batch)
        else:
            got = {d.get("code") for d in items}
            for c in batch:
                if c not in got:
                    empty_codes.append(c)
            rows = []
            for d in items:
                code = d.get("code")
                if not code or code not in w2ts:
                    continue
                end = str(d.get("披露截止") or "").replace("-", "")
                if len(end) != 8:
                    continue
                cap = d.get("持股市值")
                ratio = d.get("持股比例(%)")
                name = d.get("name")
                rows.append((w2ts[code], end, name, cap, ratio))
                ok_codes.add(code)
            for ts_code, tdate, name, cap, ratio in rows:
                if (ts_code, tdate) in existing:
                    cur.execute(
                        "UPDATE northbound_holding SET name=?, holding_market_cap=?, "
                        "holding_ratio=? WHERE ts_code=? AND trade_date=?",
                        (name, cap, ratio, ts_code, tdate))
                    n_update += 1
                else:
                    cur.execute(
                        "INSERT INTO northbound_holding "
                        "(trade_date, ts_code, name, close, change_pct, "
                        "holding_market_cap, holding_ratio) VALUES (?,?,?,?,?,?,?)",
                        (tdate, ts_code, name, None, None, cap, ratio))
                    n_insert += 1
                    existing.add((ts_code, tdate))
            conn.commit()
            print(f"[batch {bi}/{len(batches)}] ok: +{len(rows)} rows "
                  f"(ins={n_insert}, upd={n_update})", flush=True)
        if bi < len(batches):
            time.sleep(SLEEP)

    cur.execute("SELECT MAX(trade_date) FROM northbound_holding")
    max_date = cur.fetchone()[0]
    cur.execute("SELECT COUNT(*) FROM northbound_holding")
    total = cur.fetchone()[0]
    conn.close()

    print("\n===== SUMMARY =====")
    print(f"codes ok (data returned): {len(ok_codes)}")
    print(f"codes empty (no data):    {len(empty_codes)}")
    print(f"codes failed (errors):    {len(fail_codes)}")
    print(f"rows inserted: {n_insert}")
    print(f"rows updated:  {n_update}")
    print(f"table total rows: {total}")
    print(f"latest trade_date: {max_date}")
    if empty_codes:
        print("empty codes:", ",".join(empty_codes))
    if fail_codes:
        print("failed codes:", fail_codes[:20])

    with open("nb_update_result.json", "w", encoding="utf-8") as f:
        json.dump({"ok": sorted(ok_codes), "empty": empty_codes,
                   "failed": [c for c, _ in fail_codes],
                   "inserted": n_insert, "updated": n_update,
                   "max_trade_date": max_date}, f, ensure_ascii=False, indent=1)


if __name__ == "__main__":
    sys.exit(main())
